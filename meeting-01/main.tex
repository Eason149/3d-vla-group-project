\begin{frame}{机器人策略：从观测和任务得到动作}
\vspace{0.20cm}
\begin{center}{\fontsize{21}{26}\selectfont\textbf{当前观测 $o_t$ + 任务指令 $x$}\par\vspace{0.14cm}$\Downarrow\quad \pi_\theta(a_t\mid o_t,x)\quad \Downarrow$\par\vspace{0.14cm}\textbf{下一步动作 $a_t$}}\end{center}
\vspace{0.38cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.30\textwidth}\tealbold{观测}\par RGB 图像、深度、关节状态或力传感器读数\end{column}
\begin{column}{0.30\textwidth}\tealbold{任务}\par “拿起红杯子”“打开抽屉”等自然语言\end{column}
\begin{column}{0.30\textwidth}\tealbold{动作}\par 末端位置、旋转、夹爪开合或关节控制量\end{column}
\end{columns}
\end{frame}

\setsource{Ref.: [1] A. Brohan \emph{et al.}, ``RT-2: Vision-Language-Action Models Transfer Web Knowledge to Robotic Control,'' in \emph{Proc. CoRL}, 2023, pp. 2165--2183.}
\begin{frame}{VLA 的三个字母分别解决什么问题}
\vspace{0.22cm}
\renewcommand{\arraystretch}{1.48}
\setlength{\tabcolsep}{5pt}
\begin{tabular}{>{\bfseries\color{Teal}}p{0.12\textwidth} p{0.21\textwidth} p{0.29\textwidth} p{0.27\textwidth}}
\toprule & 输入或输出 & 要回答的问题 & 例子 \\ \midrule
Vision & 图像或视频 & 场景中有什么，物体在哪里 & 红杯在桌面左侧 \\
Language & 文本指令 & 人希望机器人完成什么 & 把红杯放进水槽 \\
Action & 控制序列 & 机械臂下一步如何运动 & 移动、旋转、闭合夹爪 \\ \bottomrule
\end{tabular}
\vspace{0.45cm}
\begin{center}\large VLA 希望用一个模型把\tealbold{感知、语义和控制}连接起来。\end{center}
\end{frame}

\setsource{Refs.: [1] A. Brohan \emph{et al.}, ``RT-1: Robotics Transformer for Real-World Control at Scale,'' arXiv:2212.06817, 2022. [2] A. Brohan \emph{et al.}, \emph{Proc. CoRL}, 2023.}
\begin{frame}{VLA 通常从机器人演示中学习}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.55\textwidth}
\textbf{训练数据是一段段轨迹}
\[\tau=(o_1,x,a_1,\;o_2,x,a_2,\ldots,o_T,x,a_T)\]
模型学习在给定观测和指令时，预测演示者采取的动作：
\[\min_\theta\; -\sum_t \log \pi_\theta(a_t\mid o_t,x)\]
\end{column}
\begin{column}{0.39\textwidth}
\textbf{直观理解}
\begin{itemize}\item 人先示范如何完成任务\item 模型模仿每个时刻的动作\item 部署时根据新观测继续预测\end{itemize}
\vspace{0.24cm}\orangebold{难点：}机器人一旦偏离演示轨迹，后续观测也会变得陌生。
\end{column}
\end{columns}
\end{frame}

\setsource{Ref.: [1] A. Brohan \emph{et al.}, ``RT-2: Vision-Language-Action Models Transfer Web Knowledge to Robotic Control,'' in \emph{Proc. CoRL}, 2023, pp. 2165--2183.}
\begin{frame}{RT-2 的关键思想：把动作也写成 token}
\vspace{0.16cm}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.56\textwidth}
{\fontsize{19}{24}\selectfont\begin{center}图像 + 指令\vspace{0.12cm}\par $\Downarrow$\vspace{0.12cm}\par 语言模型生成动作 token\end{center}}
\vspace{0.20cm}动作可以离散成一串编号，例如位置、旋转和夹爪状态各自对应若干 token。
\end{column}
\begin{column}{0.38\textwidth}
\textbf{这样做的好处}
\begin{itemize}\item 复用视觉语言模型的知识\item 文本与机器人数据可以共同训练\item 一个接口同时输出语言和动作\end{itemize}
\vspace{0.20cm}\orangebold{代价：}离散化会损失连续控制精度。
\end{column}
\end{columns}
\end{frame}

\setsource{Refs.: [1] H. Zhen \emph{et al.}, \emph{Proc. ICML}, 2024. [2] Y. Hong \emph{et al.}, ``3D-LLM: Injecting the 3D World into Large Language Models,'' arXiv:2307.12981, 2023.}
\begin{frame}{为什么仅看 2D 图像还不够}
\vspace{0.18cm}
\renewcommand{\arraystretch}{1.42}
\begin{tabular}{>{\bfseries}p{0.20\textwidth} p{0.34\textwidth} p{0.36\textwidth}}
\toprule 挑战 & 2D 图像中的困难 & 控制需要的信息 \\ \midrule
距离 & 同样大小的物体可能远近不同 & 抓取点与机械臂之间的真实距离 \\
遮挡 & 看不到被挡住的几何结构 & 物体是否可抓、路径是否会碰撞 \\
视角变化 & 像素位置随相机移动而变化 & 相对稳定的三维坐标 \\
空间语言 & “最远”“上方”“抽屉里面”含义模糊 & 可计算的三维关系 \\ \bottomrule
\end{tabular}
\vspace{0.42cm}
\begin{center}\large\tealbold{机器人在三维世界中行动，因此空间表征必须服务于控制。}\end{center}
\end{frame}

\setsource{Ref.: [1] H. Zhen \emph{et al.}, ``3D-VLA: A 3D Vision-Language-Action Generative World Model,'' in \emph{Proc. ICML}, vol. 235, 2024, Secs. 3.2 and 4.2.2, Fig. 6.}
\begin{frame}{三个常见的 3D 表征}
\begin{columns}[c,onlytextwidth]
\begin{column}{0.46\textwidth}\includegraphics[page=17,trim=28 390 28 22,clip,width=\linewidth]{../references/zhen24a-3d-vla.pdf}\end{column}
\begin{column}{0.48\textwidth}
\textbf{深度图 RGB-D}\par 每个像素除了颜色，还带有到相机的距离。\vspace{0.22cm}
\textbf{点云 Point Cloud}\par 把像素投影到三维坐标，得到很多 $(x,y,z)$ 点。\vspace{0.22cm}
\textbf{3D 包围盒}\par 用一个长方体近似物体的位置和尺寸，便于定位目标。
\end{column}
\end{columns}
\end{frame}

\setsource{Refs.: [1] H. Zhen \emph{et al.}, \emph{Proc. ICML}, 2024. [2] D. Ha and J. Schmidhuber, ``World Models,'' arXiv:1803.10122, 2018.}
\begin{frame}{世界模型：预测“如果这样做，会发生什么”}
\vspace{0.12cm}
\begin{center}{\fontsize{20}{25}\selectfont 当前状态 $s_t$ + 候选动作 $a_t$\quad$\longrightarrow$\quad 预测状态 $\hat{s}_{t+1}$}\end{center}
\vspace{0.42cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.45\textwidth}\textbf{普通策略}\par 直接从当前观测输出动作。模型可能学会操作，但内部依据难以观察。\end{column}
\begin{column}{0.45\textwidth}\textbf{3D-VLA 的做法}\par 先生成任务完成后的目标图像或点云，再用目标状态帮助动作预测。\end{column}
\end{columns}
\vspace{0.42cm}
\begin{center}\large\tealbold{“想象目标状态”成为语言意图与动作之间的中间变量。}\end{center}
\end{frame}

\setsource{Ref.: [1] H. Zhen \emph{et al.}, ``3D-VLA: A 3D Vision-Language-Action Generative World Model,'' in \emph{Proc. ICML}, vol. 235, 2024.}
\begin{frame}{3D-VLA 在普通 VLA 上增加了什么}
\vspace{0.24cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.29\textwidth}{\color{Teal}\Large\textbf{01}}\par\textbf{三维场景特征}\par\vspace{0.10cm}多视角图像、深度和点云提供空间信息。\end{column}
\begin{column}{0.29\textwidth}{\color{Teal}\Large\textbf{02}}\par\textbf{交互 token}\par\vspace{0.10cm}统一表示场景、物体位置、目标模态和机器人动作。\end{column}
\begin{column}{0.34\textwidth}{\color{Teal}\Large\textbf{03}}\par\textbf{生成式目标状态}\par\vspace{0.10cm}扩散模型生成目标 RGB-D 或点云，辅助规划。\end{column}
\end{columns}
\vspace{0.65cm}
\begin{center}{\large\textbf{论文的一句话主张}}\par\vspace{0.12cm}机器人如果理解三维场景，并能想象任务完成后的状态，就能更好地推理和规划动作。\end{center}
\end{frame}

\begin{frame}[fragile]{交互 token：让语言模型读写机器人变量}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.46\textwidth}
\textbf{场景和物体}
\begin{itemize}\item \texttt{<scene> ... </scene>}\item \texttt{<obj> apple </obj>}\item \texttt{<loc0-255>} 表示 3D 位置\end{itemize}
\vspace{0.20cm}\textbf{机器人动作}
\begin{itemize}\item \texttt{<aloc0-255>}：位置\item \texttt{<arot0-255>}：旋转\item \texttt{<gripper0/1>}：夹爪\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{示例输出}\par\vspace{0.10cm}
{\small\ttfamily <img> pick up\\<obj> chip bag </obj>\\<loc...> </img>}\vspace{0.42cm}
\textbf{为什么有用}\par 同一个模型接口可以处理语义、三维位置、目标模态和动作。\vspace{0.22cm}
\orangebold{为什么有风险}\par 连续空间被量化成有限 token，精细控制会受影响。
\end{column}
\end{columns}
\end{frame}

\vspace{0.20cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.30\textwidth}{\color{Teal}\Large\textbf{01}}\par\textbf{训练目标生成器}\par\vspace{0.12cm}在机器人数据上训练 RGB-D 编辑与点云扩散模型。\end{column}

\begin{frame}{结论、局限与下一步}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{今天应带走的结论}
\begin{enumerate}\item VLA 把视觉、语言和机器人动作放进一个模型。\item 3D 表征提供距离、位置和几何关系。\item 3D-VLA 先生成目标状态，再辅助动作预测。\end{enumerate}
\end{column}
\begin{column}{0.45\textwidth}
\textbf{论文仍未解决}
\begin{itemize}\item 小物体的精确控制\item 生成模型的物体消失与错误变化\item 真实深度噪声和尺度不一致\item 长序列闭环执行与开放世界泛化\end{itemize}
\end{column}
\end{columns}
\vspace{0.32cm}
\begin{center}\large\tealbold{建议起点：先运行目标图像生成 demo，再比较有无目标状态的动作预测。}\end{center}
\end{frame}
